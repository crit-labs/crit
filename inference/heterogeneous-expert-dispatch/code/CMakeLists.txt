cmake_minimum_required(VERSION 3.28)
project(crit_fused_moe LANGUAGES CXX CUDA)
if(NOT TARGET crit_fp8)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../quantized-projections/code" "${CMAKE_CURRENT_BINARY_DIR}/fp8")
endif()
add_library(crit_fused_moe STATIC fused_moe.cu)
target_compile_features(crit_fused_moe PUBLIC cxx_std_20 cuda_std_20)
target_include_directories(crit_fused_moe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(crit_fused_moe PUBLIC crit_fp8)
target_compile_options(crit_fused_moe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
