function(crit_strict_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    # -Wshadow catches the local, parameter and member hiding that MSVC /W4 rejects (C4456-C4458). For C4459 (a
    # local hiding a namespace-scope variable) GCC misses variables in anonymous namespaces, which only Clang
    # reports (CI also builds Core with Clang on Linux), and neither reports a variable of an enclosing named
    # namespace seen from a nested one.
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
  endif()
endfunction()

file(READ ${CRIT_CONFIG_KEYS_JSON} crit_config_keys)
set(crit_config_keys_inc "// Generated by core/CMakeLists.txt from python/crit_core/model/config_keys.json; edit that file.\n")
foreach(field text_sections position_lengths norm_eps dtype experts layer_types layer_count_from)
  string(JSON count LENGTH "${crit_config_keys}" ${field})
  math(EXPR last "${count} - 1")
  set(items "")
  foreach(i RANGE ${last})
    string(JSON key GET "${crit_config_keys}" ${field} ${i})
    string(APPEND items "\"${key}\", ")
  endforeach()
  string(APPEND crit_config_keys_inc "constexpr const char* ${field}_keys[] = {${items}};\n")
endforeach()
