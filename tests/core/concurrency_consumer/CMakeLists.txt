cmake_minimum_required(VERSION 3.28)
project(crit_concurrency_consumer LANGUAGES CXX)
find_package(critCore 0.6.0 EXACT CONFIG REQUIRED)
add_executable(crit_concurrency_consumer main.cpp)
target_link_libraries(crit_concurrency_consumer PRIVATE crit::core)
if(MSVC)
  target_compile_options(crit_concurrency_consumer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(crit_concurrency_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(crit_frontend_consumer ../frontend_tests.cpp)
target_link_libraries(crit_frontend_consumer PRIVATE crit::core)
# Every installed header, compiled as ordinary (not system) includes under strict warnings, with one call per
# component. The checkout's core/include/crit headers, the installed crit headers and the includes in
# installed_headers.cpp must be the same set, so a public header missing from Core's install(FILES ...), or one
# installed without updating this consumer, fails here. The cuda*.h headers are installed only with CRIT_CORE_CUDA and
# are built by the CUDA SDK consumer.
get_filename_component(_crit_public_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../../core/include/crit" ABSOLUTE)
if(NOT IS_DIRECTORY "${_crit_public_dir}")
  message(FATAL_ERROR "this consumer builds from a crit checkout: ${_crit_public_dir} is missing")
endif()
get_target_property(_crit_includes crit::core INTERFACE_INCLUDE_DIRECTORIES)
list(GET _crit_includes 0 _crit_include)
file(GLOB_RECURSE _crit_public RELATIVE "${_crit_public_dir}" CONFIGURE_DEPENDS "${_crit_public_dir}/*.h")
file(GLOB_RECURSE _crit_installed RELATIVE "${_crit_include}/crit" "${_crit_include}/crit/*.h")
set(_crit_source "${CMAKE_CURRENT_SOURCE_DIR}/installed_headers.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_crit_source}")
file(STRINGS "${_crit_source}" _crit_included REGEX "^#include <crit/[a-z0-9_/]+\\.h>")
list(TRANSFORM _crit_included REPLACE "^#include <crit/([a-z0-9_/]+\\.h)>.*$" "\\1")
foreach(_crit_list IN ITEMS _crit_public _crit_installed)
  list(FILTER ${_crit_list} EXCLUDE REGEX "^cuda.*\\.h$")
  list(SORT ${_crit_list})
endforeach()
list(SORT _crit_included)
set(_crit_not_installed ${_crit_public})
list(REMOVE_ITEM _crit_not_installed ${_crit_installed})
if(_crit_not_installed)
  message(FATAL_ERROR "core/include/crit headers missing from Core's install(FILES ...): ${_crit_not_installed}")
endif()
if(NOT _crit_installed STREQUAL _crit_public)
  message(FATAL_ERROR "installed crit headers differ from core/include/crit (stale install prefix?): "
    "installed [${_crit_installed}], core/include/crit [${_crit_public}]")
endif()
if(NOT _crit_installed STREQUAL _crit_included)
  message(FATAL_ERROR "installed_headers.cpp must include every installed crit header: "
    "installed [${_crit_installed}], included [${_crit_included}]")
endif()
add_executable(crit_installed_headers installed_headers.cpp)
target_link_libraries(crit_installed_headers PRIVATE crit::core)
set_target_properties(crit_installed_headers PROPERTIES NO_SYSTEM_FROM_IMPORTED ON)
if(MSVC)
  target_compile_options(crit_installed_headers PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(crit_installed_headers PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
include(CTest)
add_test(NAME installed_concurrent_consumer COMMAND crit_concurrency_consumer)
add_test(NAME installed_frontend_consumer COMMAND crit_frontend_consumer)
add_test(NAME installed_headers_consumer COMMAND crit_installed_headers)
set_tests_properties(installed_concurrent_consumer installed_frontend_consumer installed_headers_consumer
  PROPERTIES TIMEOUT 30)
