cmake_minimum_required(VERSION 3.20)
project(crit_kernels_openblas LANGUAGES CXX)
function(kernels_warnings target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_executable(openblas_check check.cpp)
kernels_warnings(openblas_check)
target_link_libraries(openblas_check PRIVATE ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  find_package(Threads REQUIRED)
  if(NOT TARGET crit::core)   # the attention check runs on Core's StateCache
    add_subdirectory(../../../core "${CMAKE_CURRENT_BINARY_DIR}/core")
  endif()
  add_executable(kernels_attention_check ../../../tests/inference/attention_tests.cpp)
  target_include_directories(kernels_attention_check PRIVATE . ../../../inference/attention-state/code)
  target_link_libraries(kernels_attention_check PRIVATE crit::core Threads::Threads ${CMAKE_DL_LIBS})
  kernels_warnings(kernels_attention_check)
  foreach(bits 32 64)
    add_library(kernels_blas_fixture${bits} SHARED ../../../tests/kernels/openblas_fixture.cpp)
    target_compile_definitions(kernels_blas_fixture${bits} PRIVATE BLAS_BITS=${bits})
    kernels_warnings(kernels_blas_fixture${bits})
    add_test(NAME kernels_blas_arithmetic_${bits} COMMAND openblas_check $<TARGET_FILE:kernels_blas_fixture${bits}>
      ${bits} fixture_sgemm fixture_config fixture_threads)
    add_test(NAME kernels_blas_pipeline_${bits} COMMAND ${Python3_EXECUTABLE} -B
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/kernels/openblas_pipeline.py
      $<TARGET_FILE:kernels_blas_fixture${bits}> $<TARGET_FILE:openblas_check> ${bits})
    set_tests_properties(kernels_blas_pipeline_${bits} PROPERTIES TIMEOUT 90)
    add_test(NAME kernels_attention_pipeline_${bits} COMMAND ${Python3_EXECUTABLE} -B
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/kernels/openblas_pipeline.py
      $<TARGET_FILE:kernels_blas_fixture${bits}> $<TARGET_FILE:kernels_attention_check> ${bits} attention)
    set_tests_properties(kernels_attention_pipeline_${bits} PROPERTIES TIMEOUT 90)
  endforeach()
  add_executable(kernels_blas_boundaries ../../../tests/kernels/openblas_boundaries.cpp)
  target_include_directories(kernels_blas_boundaries PRIVATE .)
  target_link_libraries(kernels_blas_boundaries PRIVATE ${CMAKE_DL_LIBS})
  kernels_warnings(kernels_blas_boundaries)
  add_test(NAME kernels_blas_boundaries COMMAND kernels_blas_boundaries
    $<TARGET_FILE:kernels_blas_fixture32> $<TARGET_FILE:kernels_blas_fixture64>)
endif()
