function(crit_strict_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    # -Wshadow catches the local, parameter and member hiding that MSVC /W4 rejects (C4456-C4458). For C4459 (a
    # local hiding a namespace-scope variable) GCC misses variables in anonymous namespaces, which only Clang
    # reports (CI also builds Core with Clang on Linux), and neither reports a variable of an enclosing named
    # namespace seen from a nested one.
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
  endif()
endfunction()

# The settings of a step engine's "engine" object (python/crit_core/step_settings.json), which crit_core.native's
# StepEngine also reads, as the C++ list of src/runtime/step_engine.cpp.
set(CRIT_STEP_SETTINGS_JSON ${CMAKE_CURRENT_SOURCE_DIR}/python/crit_core/step_settings.json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CRIT_STEP_SETTINGS_JSON})
file(READ ${CRIT_STEP_SETTINGS_JSON} crit_step_settings)
string(JSON count LENGTH "${crit_step_settings}" settings)
math(EXPR last "${count} - 1")
set(items "")
foreach(i RANGE ${last})
  string(JSON key GET "${crit_step_settings}" settings ${i})
  string(APPEND items "\"${key}\", ")
endforeach()
set(crit_step_settings_inc "// Generated by core/CMakeLists.txt from python/crit_core/step_settings.json; edit that file.\n")
string(APPEND crit_step_settings_inc "constexpr std::array<std::string_view, ${count}> step_settings{${items}};\n")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/crit_step_settings.inc CONTENT "${crit_step_settings_inc}" @ONLY)
