cmake_minimum_required(VERSION 3.28)
project(crit_recurrent_recovery LANGUAGES CXX)
include(CTest)
# checkpoint_node comes from Core (crit/state.h): an installed SDK, else this checkout's core/.
if(NOT TARGET crit::core)
  find_package(critCore 0.10.0 CONFIG QUIET)
  if(NOT critCore_FOUND)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../core" "${CMAKE_CURRENT_BINARY_DIR}/core")
  endif()
endif()
add_executable(crit_recurrent_recovery_example example.cpp)
target_compile_features(crit_recurrent_recovery_example PRIVATE cxx_std_20)
target_link_libraries(crit_recurrent_recovery_example PRIVATE crit::core)
if(BUILD_TESTING)
  add_executable(crit_recurrent_recovery_tests ../../../tests/inference/recurrent_recovery_tests.cpp)
  target_include_directories(crit_recurrent_recovery_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(crit_recurrent_recovery_tests PRIVATE cxx_std_20)
  target_link_libraries(crit_recurrent_recovery_tests PRIVATE crit::core)
  add_test(NAME crit_recurrent_recovery COMMAND crit_recurrent_recovery_tests)
  add_test(NAME crit_recurrent_recovery_example COMMAND crit_recurrent_recovery_example)
endif()
foreach(target crit_recurrent_recovery_example crit_recurrent_recovery_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
endforeach()
