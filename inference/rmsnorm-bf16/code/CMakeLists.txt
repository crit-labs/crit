cmake_minimum_required(VERSION 3.28)
project(crit_rmsnorm LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(crit_rmsnorm STATIC interface.cpp implementation.cu)
add_library(crit::rmsnorm ALIAS crit_rmsnorm)
target_compile_features(crit_rmsnorm PUBLIC cxx_std_20 cuda_std_20)
target_include_directories(crit_rmsnorm PUBLIC include)
target_link_libraries(crit_rmsnorm PUBLIC CUDA::cudart)
option(CRIT_RMSNORM_EXAMPLE "Build the standalone BF16 borrowed-view example" OFF)
if(CRIT_RMSNORM_EXAMPLE)
  add_executable(crit_rmsnorm_example example.cpp)
  target_link_libraries(crit_rmsnorm_example PRIVATE crit::rmsnorm)
endif()
